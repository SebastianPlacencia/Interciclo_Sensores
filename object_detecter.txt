#!/usr/bin/env python3
import os
import time
os.environ["QT_QPA_PLATFORM"] = "xcb"
os.environ["LIBGL_ALWAYS_SOFTWARE"] = "1"
os.environ["XDG_RUNTIME_DIR"] = "/tmp"

import rclpy
from rclpy.node import Node
from sensor_msgs.msg import Image
from std_msgs.msg import Int32
from cv_bridge import CvBridge
import cv2
import numpy as np

class ObjectDetector(Node):
    def __init__(self):
        super().__init__('object_detector')
        self.br = CvBridge()
        self.subscription = self.create_subscription(Image, '/tello_image', self.image_callback, 10)
        
        # PUBLICADORES ESTRICTOS TIPO INT32 SEGUN LA GUIA
        self.red_pub = self.create_publisher(Int32, '/red_object_count', 10)
        self.black_pub = self.create_publisher(Int32, '/black_object_count', 10)
        
        self.frame_counter = 0 
        self.prev_frame_time = time.time()
        self.get_logger().info('Detector Visual Iniciado. Calculando Metricas de Rendimiento.')

    def count_contours(self, mask, frame, color_bgr, label):
        mask = cv2.erode(mask, None, iterations=1)
        mask = cv2.dilate(mask, None, iterations=1)
        contours, _ = cv2.findContours(mask, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_SIMPLE)
        count = 0
        for cnt in contours:
            if cv2.contourArea(cnt) > 300: 
                count += 1
                x, y, w, h = cv2.boundingRect(cnt)
                cv2.rectangle(frame, (x, y), (x + w, y + h), color_bgr, 2)
                cv2.putText(frame, label, (x, y - 5), cv2.FONT_HERSHEY_SIMPLEX, 0.6, color_bgr, 2)
        return count

    def image_callback(self, msg):
        try:
            self.frame_counter += 1
            if self.frame_counter % 2 != 0: return # Frame Skipping

            # Inicio del cronometro para medir Latencia de Procesamiento
            start_process_time = time.time()

            frame = self.br.imgmsg_to_cv2(msg, "bgr8")
            frame = cv2.cvtColor(frame, cv2.COLOR_RGB2BGR)
            frame = cv2.resize(frame, (640, 480))
            hsv = cv2.cvtColor(frame, cv2.COLOR_BGR2HSV)

            low_red1, high_red1 = np.array([0, 100, 100]), np.array([10, 255, 255])
            low_red2, high_red2 = np.array([160, 100, 100]), np.array([180, 255, 255])
            low_black, high_black = np.array([0, 0, 0]), np.array([180, 45, 45])

            mask_red = cv2.inRange(hsv, low_red1, high_red1) + cv2.inRange(hsv, low_red2, high_red2)
            mask_black = cv2.inRange(hsv, low_black, high_black)

            current_red = self.count_contours(mask_red, frame, (0, 0, 255), "ROJO")
            current_black = self.count_contours(mask_black, frame, (0, 0, 0), "NEGRO")

            # PUBLICACION DE TOPICOS INT32
            msg_red = Int32()
            msg_red.data = current_red
            self.red_pub.publish(msg_red)

            msg_black = Int32()
            msg_black.data = current_black
            self.black_pub.publish(msg_black)

            # Fin del cronometro: Calculo de Latencia y FPS
            end_process_time = time.time()
            processing_latency = (end_process_time - start_process_time) * 1000 # En milisegundos
            
            fps = 1.0 / (end_process_time - self.prev_frame_time) if (end_process_time - self.prev_frame_time) > 0 else 0
            self.prev_frame_time = end_process_time

            # Mostrar datos y metricas en pantalla
            cv2.rectangle(frame, (10, 10), (220, 130), (255, 255, 255), -1)
            cv2.putText(frame, f"Roj: {current_red}", (20, 35), cv2.FONT_HERSHEY_SIMPLEX, 0.7, (0, 0, 255), 2)
            cv2.putText(frame, f"Neg: {current_black}", (20, 65), cv2.FONT_HERSHEY_SIMPLEX, 0.7, (0, 0, 0), 2)
            cv2.putText(frame, f"FPS: {int(fps)}", (20, 95), cv2.FONT_HERSHEY_SIMPLEX, 0.6, (150, 0, 0), 2)
            cv2.putText(frame, f"Lat: {processing_latency:.1f} ms", (20, 120), cv2.FONT_HERSHEY_SIMPLEX, 0.6, (150, 0, 0), 2)

            cv2.imshow("Vision Tello - Metricas", frame)
            cv2.waitKey(1)
        except Exception as e:
            self.get_logger().error(f'Error: {e}')

def main(args=None):
    rclpy.init(args=args)
    node = ObjectDetector()
    try: rclpy.spin(node)
    except KeyboardInterrupt: pass
    finally:
        cv2.destroyAllWindows()
        node.destroy_node()
        rclpy.shutdown()

if __name__ == '__main__': main()
